"""Battery Energy Storage System (BESS) physical model for microgrid_sim.

Governing Physical Equations:
    1. Sign Convention:
       - In signed power representation P_batt:
         P_batt > 0: Discharging (acting as a source, injecting power to microgrid).
         P_batt < 0: Charging (acting as a sink, absorbing power from microgrid).
         P_batt = 0: Idle.
       - In dual-variable representation (P_ch, P_dis):
         P_ch >= 0 (charging power), P_dis >= 0 (discharging power).
         Mutual exclusivity: P_ch * P_dis = 0.

    2. Dynamic Feasibility Limits (SoC Envelope Protection):
       - Maximum feasible charge power over interval dt without exceeding SoC_max:
         P_ch_max_dyn = min(P_ch_max, [(SoC_max - SoC * (1 - sigma_sd * dt)) * E_nom] / (eta_ch * dt))
       - Maximum feasible discharge power over interval dt without breaching SoC_min:
         P_dis_max_dyn = min(P_dis_max, [(SoC * (1 - sigma_sd * dt) - SoC_min) * E_nom * eta_dis] / dt)

    3. Energy Conservation & Conversion Losses:
       - Energy absorbed during charging: E_terminal_ch = P_ch * dt [kWh]
         Internal cell energy stored:     E_cell_ch = E_terminal_ch * eta_ch [kWh]
         Conversion loss:                 E_loss_ch = E_terminal_ch * (1 - eta_ch) [kWh]
       - Energy delivered during discharge: E_terminal_dis = P_dis * dt [kWh]
         Internal cell energy extracted:    E_cell_dis = E_terminal_dis / eta_dis [kWh]
         Conversion loss:                   E_loss_dis = E_cell_dis - E_terminal_dis [kWh]
       - Self-discharge loss:
         E_loss_sd = SoC * E_nom * sigma_sd * dt [kWh]

    4. State of Charge Evolution:
       SoC(t + dt) = SoC(t) * (1 - sigma_sd * dt) + (E_cell_ch - E_cell_dis) / E_nom

    5. Energy Balance Invariant on Executed Power:
       E_terminal_executed = Delta_E_stored + E_losses_total
       Unexecutable requested energy is tracked explicitly as E_rejected and is NOT absorbed or created.
"""

from dataclasses import dataclass
from typing import Optional, Tuple
import numpy as np

from microgrid_sim.config.schemas import BatteryConfig
from microgrid_sim.utils.exceptions import PhysicsViolationError
from microgrid_sim.utils.logger import get_logger

logger = get_logger(__name__)


@dataclass(frozen=True)
class BatteryStepResult:
    """Detailed physical outputs and telemetry from a single battery simulation step."""
    requested_power_kw: float
    executed_power_kw: float
    rejected_power_kw: float
    energy_rejected_kwh: float
    soc_before: float
    soc_after: float
    energy_charged_kwh: float
    energy_discharged_kwh: float
    internal_energy_stored_kwh: float
    internal_energy_extracted_kwh: float
    losses_kwh: float
    constraint_encountered: str
    wear_cost_usd: float


class BatteryModel:
    """Physics-based Battery Energy Storage System (BESS) model.

    Enforces dynamic SoC limits, conversion efficiencies, self-discharge,
    and mechanical/economic wear accounting.
    """

    def __init__(self, config: BatteryConfig, timestep_hours: float = 0.25) -> None:
        """Initialize the BESS model with validated configuration.

        Args:
            config: Validated BatteryConfig schema instance.
            timestep_hours: Simulation timestep duration [hours] (default: 0.25 h / 15 min).
        """
        self.config = config
        self.dt = timestep_hours

        # Capacity & SoC boundaries
        self.nominal_energy_kwh = config.capacity.nominal_energy_kwh
        self.min_soc = config.capacity.minimum_soc
        self.max_soc = config.capacity.maximum_soc
        self.soc = config.capacity.initial_soc

        # Power ratings
        self.max_p_ch = config.power_and_thermal.max_charge_power_kw
        self.max_p_dis = config.power_and_thermal.max_discharge_power_kw

        # Efficiencies & Self-Discharge
        self.eta_ch = config.performance.charging_efficiency
        self.eta_dis = config.performance.discharging_efficiency
        self.sigma_sd = config.performance.self_discharge_rate_per_hour

        # Degradation parameters
        self.wear_cost_per_kwh = config.degradation.implied_wear_cost_usd_per_kwh
        self.cycle_life_to_eol = config.degradation.cycle_life_to_eol

        # Cumulative tracking
        self.cumulative_energy_charged_kwh = 0.0
        self.cumulative_energy_discharged_kwh = 0.0
        self.cumulative_losses_kwh = 0.0
        self.cumulative_wear_cost_usd = 0.0

        logger.debug(
            "Initialized BatteryModel: %.1f kWh capacity, SoC in [%.2f, %.2f], Initial SoC: %.2f",
            self.nominal_energy_kwh, self.min_soc, self.max_soc, self.soc
        )

    def get_dynamic_power_limits(self, current_soc: float) -> Tuple[float, float]:
        """Compute maximum allowable charging and discharging power over interval dt.

        Ensures that applying the returned power limits strictly preserves the
        configured operational SoC envelope [min_soc, max_soc].

        Args:
            current_soc: Current battery State of Charge [0.0, 1.0].

        Returns:
            Tuple of (p_ch_max_dyn, p_dis_max_dyn) in [kW].
        """
        # Account for self-discharge over the timestep
        soc_decayed = current_soc * (1.0 - self.sigma_sd * self.dt)

        # Max charge power limited by max_soc and charge power rating
        headroom_soc = max(0.0, self.max_soc - soc_decayed)
        p_ch_dyn = (headroom_soc * self.nominal_energy_kwh) / (self.eta_ch * self.dt)
        p_ch_max_dyn = min(self.max_p_ch, p_ch_dyn)

        # Max discharge power limited by min_soc and discharge power rating
        available_soc = max(0.0, soc_decayed - self.min_soc)
        p_dis_dyn = (available_soc * self.nominal_energy_kwh * self.eta_dis) / self.dt
        p_dis_max_dyn = min(self.max_p_dis, p_dis_dyn)

        return float(max(0.0, p_ch_max_dyn)), float(max(0.0, p_dis_max_dyn))

    def step(self, requested_power_kw: float) -> BatteryStepResult:
        """Execute battery simulation step using signed power convention.

        Convention:
            requested_power_kw > 0: Discharging request (delivering power to microgrid).
            requested_power_kw < 0: Charging request (absorbing power from microgrid).
            requested_power_kw == 0: Idle request.

        Args:
            requested_power_kw: Requested battery active power [kW].

        Returns:
            BatteryStepResult dataclass containing execution details.
        """
        soc_before = self.soc
        p_ch_dyn, p_dis_dyn = self.get_dynamic_power_limits(soc_before)

        constraint = "NONE"
        actual_p_ch = 0.0
        actual_p_dis = 0.0

        if requested_power_kw < 0.0:
            # Charging request: requested_power_kw is negative
            req_ch = abs(requested_power_kw)
            if req_ch > self.max_p_ch:
                constraint = "MAX_CHARGE_POWER"
            actual_p_ch = min(req_ch, p_ch_dyn)
            if actual_p_ch < req_ch and constraint == "NONE":
                constraint = "MAX_SOC"
            executed_power = -actual_p_ch

        elif requested_power_kw > 0.0:
            # Discharging request: requested_power_kw is positive
            req_dis = requested_power_kw
            if req_dis > self.max_p_dis:
                constraint = "MAX_DISCHARGE_POWER"
            actual_p_dis = min(req_dis, p_dis_dyn)
            if actual_p_dis < req_dis and constraint == "NONE":
                constraint = "MIN_SOC"
            executed_power = actual_p_dis

        else:
            executed_power = 0.0

        # Rejected unexecuted power and energy (strictly non-negative magnitude)
        rejected_power_kw = max(0.0, abs(requested_power_kw) - abs(executed_power))
        energy_rejected_kwh = float(rejected_power_kw * self.dt)

        # Energy quantities over timestep dt
        e_terminal_ch = actual_p_ch * self.dt
        e_terminal_dis = actual_p_dis * self.dt

        e_cell_ch = e_terminal_ch * self.eta_ch
        e_cell_dis = e_terminal_dis / self.eta_dis if self.eta_dis > 0.0 else 0.0

        loss_ch = e_terminal_ch * (1.0 - self.eta_ch)
        loss_dis = e_cell_dis - e_terminal_dis
        loss_sd = soc_before * self.nominal_energy_kwh * self.sigma_sd * self.dt
        step_losses = loss_ch + loss_dis + loss_sd

        # State of charge evolution
        soc_after = soc_before * (1.0 - self.sigma_sd * self.dt) + (e_cell_ch - e_cell_dis) / self.nominal_energy_kwh
        # Numerical guard: clamp within boundaries
        soc_after = float(min(max(soc_after, self.min_soc), self.max_soc))
        self.soc = soc_after

        # Economic degradation wear
        wear_cost = (e_terminal_ch + e_terminal_dis) * self.wear_cost_per_kwh

        # Cumulative updates
        self.cumulative_energy_charged_kwh += e_terminal_ch
        self.cumulative_energy_discharged_kwh += e_terminal_dis
        self.cumulative_losses_kwh += step_losses
        self.cumulative_wear_cost_usd += wear_cost

        return BatteryStepResult(
            requested_power_kw=float(requested_power_kw),
            executed_power_kw=float(executed_power),
            rejected_power_kw=float(rejected_power_kw),
            energy_rejected_kwh=float(energy_rejected_kwh),
            soc_before=float(soc_before),
            soc_after=float(soc_after),
            energy_charged_kwh=float(e_terminal_ch),
            energy_discharged_kwh=float(e_terminal_dis),
            internal_energy_stored_kwh=float(e_cell_ch),
            internal_energy_extracted_kwh=float(e_cell_dis),
            losses_kwh=float(step_losses),
            constraint_encountered=constraint,
            wear_cost_usd=float(wear_cost),
        )

    def step_ch_dis(
        self,
        charge_power_kw: float = 0.0,
        discharge_power_kw: float = 0.0,
        **kwargs: float,
    ) -> BatteryStepResult:
        """Execute step using explicit separate charging and discharging inputs.

        Enforces mutual exclusivity (P_ch * P_dis = 0).

        Args:
            charge_power_kw: Requested charging power [kW >= 0].
            discharge_power_kw: Requested discharging power [kW >= 0].
            **kwargs: Backward-compatible keyword arguments ('p_ch_kw', 'p_dis_kw').

        Returns:
            BatteryStepResult dataclass.

        Raises:
            PhysicsViolationError: If simultaneous charge and discharge is requested, or if inputs are negative.
        """
        ch = float(kwargs.get("p_ch_kw", charge_power_kw))
        dis = float(kwargs.get("p_dis_kw", discharge_power_kw))

        if ch < 0.0 or dis < 0.0:
            raise PhysicsViolationError(
                f"Negative power inputs to step_ch_dis: charge={ch}, discharge={dis}"
            )

        if ch > 0.0 and dis > 0.0:
            raise PhysicsViolationError(
                f"Simultaneous battery charging ({ch} kW) and discharging ({dis} kW) violation"
            )

        if ch > 0.0:
            return self.step(-ch)
        elif dis > 0.0:
            return self.step(dis)
        else:
            return self.step(0.0)

"""Power Conversion System (Inverter and PCS) physical model for microgrid_sim.

Governing Equations and Scientific Classification:
    1. Solar PV Inverter (Unidirectional DC to AC):
       - Part-load ratio:
           p_pu = P_dc_accepted / P_rated_ac
       - Empirical Quadratic Part-Load Efficiency Model:
           Loss_poly = c0 + c1 * (p_pu - 1.0)^2
           eta_inv = eta_nom * (1.0 - Loss_poly)

       Scientific Note on Model Classification:
       This formulation is an Empirical Quadratic Part-Load Model, NOT the 6-parameter
       empirical Sandia Performance Model for Grid-Connected Photovoltaic Inverters
       (Sandia Report SAND2004-5221 / California Energy Commission).
       Under standard benchmark configuration (eta_nom=0.975, c0=0.015, c1=0.035):
       At rated load (p_pu = 1.0), the quadratic term vanishes, and the baseline loss
       offset c0=0.015 yields an effective efficiency of:
           eta(1.0) = 0.975 * (1.0 - 0.015) = 0.975 * 0.985 = 0.960375 (~96.04%)
       The nominal parameter eta_nom = 0.975 represents the theoretical asymptotic ceiling
       before baseline conduction and curvature deductions are evaluated.

       - Exact Energy and Power Separation:
           P_dc_clipped = max(0.0, P_dc_input - P_max_dc)
           P_dc_accepted = min(P_dc_input, P_max_dc)
           eta_inv = calculate_pv_efficiency(P_dc_accepted)
           P_ac_potential = P_dc_accepted * eta_inv
           P_loss_thermal = P_dc_accepted * (1.0 - eta_inv)
           P_ac_clipped = max(0.0, P_ac_potential - P_rated_ac)
           P_ac_output = min(P_rated_ac, P_ac_potential)
           P_curtailed = P_dc_clipped + P_ac_clipped

       - Conservation of Energy Identity:
           P_dc_input == P_ac_output + P_loss_thermal + P_curtailed

    2. Battery Power Conversion System (PCS, Bidirectional AC-DC):
       - Distinct from PV Inverter with independent electrical parameters:
           P_rated_pcs = 100.0 kW_ac, eta_pcs = 0.96
       - Discharging (DC to AC):
           P_ac_raw = P_dc_dis * eta_pcs
           P_ac_dis = min(P_rated_pcs, P_ac_raw)
           P_clipped_dis = max(0.0, P_ac_raw - P_rated_pcs)
           P_loss_dis = P_dc_dis * (1.0 - eta_pcs)
       - Charging (AC to DC):
           P_ac_ch_clipped = min(P_rated_pcs, P_ac_ch)
           P_clipped_ch = max(0.0, P_ac_ch - P_rated_pcs)
           P_dc_ch = P_ac_ch_clipped * eta_pcs
           P_loss_ch = P_ac_ch_clipped * (1.0 - eta_pcs)

    3. Apparent Power Limit (IEEE 1547):
       S = sqrt(P^2 + Q^2) <= S_max. At unity power factor (Q = 0), P <= S_max.
"""

from dataclasses import dataclass
from typing import Optional, Tuple
import numpy as np

from microgrid_sim.config.schemas import InverterConfig
from microgrid_sim.utils.exceptions import PhysicsViolationError
from microgrid_sim.utils.logger import get_logger

logger = get_logger(__name__)


@dataclass(frozen=True)
class PVInverterResult:
    """Detailed outputs from solar PV inverter conversion."""
    p_dc_input_kw: float
    p_ac_output_kw: float
    p_curtailed_kw: float
    p_loss_kw: float
    efficiency: float
    is_clipped: bool
    p_dc_clipped_kw: float = 0.0
    p_ac_clipped_kw: float = 0.0


@dataclass(frozen=True)
class BatteryPCSResult:
    """Detailed outputs from battery power conversion system (PCS)."""
    mode: str  # "CHARGING", "DISCHARGING", "IDLE"
    p_ac_kw: float
    p_dc_kw: float
    p_loss_kw: float
    efficiency: float
    is_clipped: bool
    p_clipped_kw: float = 0.0


class InverterModel:
    """Physics-based Power Conversion System for Solar PV and Battery BESS.

    Implements empirical quadratic part-load efficiency modeling for the PV inverter and
    dedicated bidirectional conversion for the battery PCS.
    """

    def __init__(self, config: InverterConfig) -> None:
        """Initialize the inverter model with validated configuration.

        Args:
            config: Validated InverterConfig schema instance.
        """
        self.config = config

        # PV Inverter ratings & empirical part-load parameters
        self.p_rated_pv_ac = config.ratings.rated_ac_power_kw
        self.s_rated_pv_kva = config.ratings.rated_apparent_power_kva
        self.max_dc_input_kw = config.ratings.max_dc_input_power_kw
        self.eta_nom = config.efficiency_model.nominal_peak_efficiency
        self.c0 = config.efficiency_model.curvature_coefficient_c0
        self.c1 = config.efficiency_model.curvature_coefficient_c1

        # Battery PCS ratings
        self.p_rated_batt_ac = config.battery_inverter.rated_ac_power_kw
        self.eta_batt_pcs = config.battery_inverter.bidirectional_efficiency

        logger.debug(
            "Initialized InverterModel: PV AC=%.1f kW (eta_nom=%.3f, rated eta=%.4f), Battery PCS AC=%.1f kW (eta=%.3f)",
            self.p_rated_pv_ac, self.eta_nom, self.eta_nom * (1.0 - self.c0), self.p_rated_batt_ac, self.eta_batt_pcs
        )

    def calculate_pv_efficiency(self, dc_power_kw: float) -> float:
        """Compute non-linear conversion efficiency for solar PV inverter.

        Uses an Empirical Quadratic Part-Load Model:
            p_pu = dc_power_kw / p_rated_pv_ac
            loss = c0 + c1 * (p_pu - 1.0)^2
            eta = eta_nom * (1.0 - loss)

        Note: At p_pu = 1.0, eta = eta_nom * (1 - c0) = 0.975 * (1 - 0.015) = 0.960375.

        Args:
            dc_power_kw: DC input power accepted by the inverter front-end [kW].

        Returns:
            Inverter efficiency [0.0, 1.0].
        """
        if dc_power_kw <= 0.0:
            return 0.0

        p_pu = dc_power_kw / self.p_rated_pv_ac
        loss = self.c0 + self.c1 * ((p_pu - 1.0) ** 2)
        eta = self.eta_nom * (1.0 - loss)
        return float(min(max(eta, 0.0), 1.0))

    def convert_pv_dc_to_ac(self, dc_power_kw: float) -> PVInverterResult:
        """Convert solar DC generation to AC grid-frequency power with explicit clipping.

        Delineates:
        1. DC-side front-end clipping: Exceeding max_dc_input_kw.
        2. Thermal dissipation loss: Conduction and switching loss P_accepted * (1 - eta).
        3. AC-side capacity clipping: Inverter thermal/apparent capacity clamp.

        Energy conservation identity:
            P_dc_input == P_ac_output + P_loss_kw + P_curtailed_kw
        where P_curtailed_kw = P_dc_clipped_kw + P_ac_clipped_kw.

        Args:
            dc_power_kw: Solar array DC output power [kW_dc >= 0].

        Returns:
            PVInverterResult dataclass with detailed power flows.
        """
        if dc_power_kw < 0.0:
            raise PhysicsViolationError(
                f"Negative DC power input {dc_power_kw} kW to PV inverter",
                details={"dc_power_kw": dc_power_kw},
            )

        if dc_power_kw == 0.0:
            return PVInverterResult(
                p_dc_input_kw=0.0,
                p_ac_output_kw=0.0,
                p_curtailed_kw=0.0,
                p_loss_kw=0.0,
                efficiency=0.0,
                is_clipped=False,
                p_dc_clipped_kw=0.0,
                p_ac_clipped_kw=0.0,
            )

        # 1. DC-side front-end clipping
        if dc_power_kw > self.max_dc_input_kw:
            p_dc_accepted = self.max_dc_input_kw
            p_dc_clipped = dc_power_kw - self.max_dc_input_kw
        else:
            p_dc_accepted = dc_power_kw
            p_dc_clipped = 0.0

        # 2. Conversion efficiency and thermal losses on accepted DC power
        eta = self.calculate_pv_efficiency(p_dc_accepted)
        p_ac_potential = p_dc_accepted * eta
        p_loss = p_dc_accepted * (1.0 - eta)

        # 3. AC-side capacity clipping at rated AC capacity (and apparent power limit at unity PF)
        ac_limit = min(self.p_rated_pv_ac, self.s_rated_pv_kva)
        if p_ac_potential > ac_limit:
            p_ac_output = ac_limit
            p_ac_clipped = p_ac_potential - ac_limit
            is_clipped = True
        else:
            p_ac_output = p_ac_potential
            p_ac_clipped = 0.0
            is_clipped = (p_dc_clipped > 0.0)

        p_curtailed = p_dc_clipped + p_ac_clipped

        return PVInverterResult(
            p_dc_input_kw=float(dc_power_kw),
            p_ac_output_kw=float(p_ac_output),
            p_curtailed_kw=float(p_curtailed),
            p_loss_kw=float(p_loss),
            efficiency=float(eta),
            is_clipped=is_clipped or (p_curtailed > 0.0),
            p_dc_clipped_kw=float(p_dc_clipped),
            p_ac_clipped_kw=float(p_ac_clipped),
        )

    def convert_battery_discharge_dc_to_ac(self, p_batt_dc_dis_kw: float) -> BatteryPCSResult:
        """Convert battery DC discharge power to AC microgrid power via battery PCS.

        Args:
            p_batt_dc_dis_kw: Battery discharging power from DC side [kW >= 0].

        Returns:
            BatteryPCSResult dataclass.
        """
        if p_batt_dc_dis_kw < 0.0:
            raise PhysicsViolationError(f"Negative DC discharge power {p_batt_dc_dis_kw} kW")

        if p_batt_dc_dis_kw == 0.0:
            return BatteryPCSResult("IDLE", 0.0, 0.0, 0.0, self.eta_batt_pcs, False, 0.0)

        p_ac_raw = p_batt_dc_dis_kw * self.eta_batt_pcs
        p_loss = p_batt_dc_dis_kw * (1.0 - self.eta_batt_pcs)

        if p_ac_raw > self.p_rated_batt_ac:
            p_ac = self.p_rated_batt_ac
            p_clipped = p_ac_raw - self.p_rated_batt_ac
            is_clipped = True
        else:
            p_ac = p_ac_raw
            p_clipped = 0.0
            is_clipped = False

        return BatteryPCSResult(
            mode="DISCHARGING",
            p_ac_kw=float(p_ac),
            p_dc_kw=float(p_batt_dc_dis_kw),
            p_loss_kw=float(p_loss),
            efficiency=float(self.eta_batt_pcs),
            is_clipped=is_clipped,
            p_clipped_kw=float(p_clipped),
        )

    def convert_battery_charge_ac_to_dc(self, p_batt_ac_ch_kw: float) -> BatteryPCSResult:
        """Convert AC microgrid power to battery DC charging power via battery PCS.

        Args:
            p_batt_ac_ch_kw: Charging power drawn from AC bus [kW >= 0].

        Returns:
            BatteryPCSResult dataclass.
        """
        if p_batt_ac_ch_kw < 0.0:
            raise PhysicsViolationError(f"Negative AC charge power {p_batt_ac_ch_kw} kW")

        if p_batt_ac_ch_kw == 0.0:
            return BatteryPCSResult("IDLE", 0.0, 0.0, 0.0, self.eta_batt_pcs, False, 0.0)

        # Clip AC input at PCS rating
        if p_batt_ac_ch_kw > self.p_rated_batt_ac:
            p_ac_actual = self.p_rated_batt_ac
            p_clipped = p_batt_ac_ch_kw - self.p_rated_batt_ac
            is_clipped = True
        else:
            p_ac_actual = p_batt_ac_ch_kw
            p_clipped = 0.0
            is_clipped = False

        p_dc = p_ac_actual * self.eta_batt_pcs
        p_loss = p_ac_actual * (1.0 - self.eta_batt_pcs)

        return BatteryPCSResult(
            mode="CHARGING",
            p_ac_kw=float(p_ac_actual),
            p_dc_kw=float(p_dc),
            p_loss_kw=float(p_loss),
            efficiency=float(self.eta_batt_pcs),
            is_clipped=is_clipped,
            p_clipped_kw=float(p_clipped),
        )

"""Microgrid AC bus power balance and conservation calculator for microgrid_sim.

Governing Physical Law (Kirchhoff Active Power Conservation on 480 V AC Bus):
    Every term in the physical power balance equation represents an actual active power flow
    injected into or extracted from the microgrid AC bus:

    Total Physical Injections (Sources) = Total Physical Extractions (Sinks)

    P_pv_ac_delivered + P_grid_import + P_batt_dis_ac + P_diesel
    =
    P_served_load + P_batt_ch_ac + P_grid_export

Where:
    - P_served_load = P_requested_load - P_unserved_load (Unserved load is unmet demand, NOT a physical source).
    - P_pv_ac_delivered = P_pv_ac_potential - P_pv_curtail_ac (Curtailed PV is generation withheld/throttled by the inverter, NOT an AC bus sink).
    - P_grid_import: Contractually bounded power imported through the PCC [kW >= 0].
    - P_grid_export: Contractually bounded surplus power exported through the PCC [kW >= 0].
    - P_batt_dis_ac: AC power delivered by battery PCS [kW >= 0].
    - P_batt_ch_ac: AC power drawn from bus to charge battery via PCS [kW >= 0].
    - P_diesel: Auxiliary backup generation [kW >= 0] (0.0 in active benchmark).

Conservation Invariant:
    Residual = |P_sources - P_sinks| <= tolerance (default: 1.0e-6 kW / 1 mW)
"""

from dataclasses import dataclass
from typing import Optional
import numpy as np

from microgrid_sim.utils.exceptions import ConservationOfEnergyError, PhysicsViolationError
from microgrid_sim.utils.logger import get_logger

@dataclass(frozen=True)
class PowerBalanceResult:
    """Detailed resolution of microgrid AC bus power balance."""
    # Physical AC Bus Flows
    p_pv_ac_delivered_kw: float
    p_grid_import_kw: float
    p_batt_dis_ac_kw: float
    p_diesel_kw: float
    p_served_load_kw: float
    p_batt_ch_ac_kw: float
    p_grid_export_kw: float

    # Slack / Operational Accounting (Demand & Solar Potential)
    p_requested_load_kw: float
    p_unserved_load_kw: float
    p_pv_ac_potential_kw: float
    p_pv_curtail_ac_kw: float

    # Verification Metrics
    residual_error_kw: float
    is_balanced: bool

    # Backwards-compatible properties
    @property
    def p_load_kw(self) -> float:
        """Alias for requested load demand [kW]."""
        return self.p_requested_load_kw

    @property
    def p_pv_ac_kw(self) -> float:
        """Alias for delivered solar PV AC power [kW]."""
        return self.p_pv_ac_delivered_kw

    @property
    def p_unserved_kw(self) -> float:
        """Alias for unserved load [kW]."""
        return self.p_unserved_load_kw

    @property
    def p_curtail_kw(self) -> float:
        """Alias for AC curtailed solar power [kW]."""
        return self.p_pv_curtail_ac_kw

    @property
    def p_sources_kw(self) -> float:
        """Total AC bus power sources (Delivered PV + Grid Import + Batt Dis + Diesel) [kW]."""
        return (
            self.p_pv_ac_delivered_kw
            + self.p_grid_import_kw
            + self.p_batt_dis_ac_kw
            + self.p_diesel_kw
        )

    @property
    def p_sinks_kw(self) -> float:
        """Total AC bus power sinks (Served Load + Batt Ch + Grid Export) [kW]."""
        return (
            self.p_served_load_kw
            + self.p_batt_ch_ac_kw
            + self.p_grid_export_kw
        )


class PowerBalanceCalculator:
    """Enforces Kirchhoff power balance and resolves physical energy flows on the microgrid AC bus."""

    def __init__(self, numerical_tolerance_kw: float = 1.0e-6) -> None:
        """Initialize the power balance calculator.

        Args:
            numerical_tolerance_kw: Maximum allowable residual power [kW] (default: 1.0e-6 kW = 1 mW).
        """
        self.tolerance_kw = numerical_tolerance_kw

    def resolve_balance(
        self,
        p_load_kw: float,
        p_pv_ac_kw: float,
        p_batt_ch_ac_kw: float,
        p_batt_dis_ac_kw: float,
        max_grid_import_kw: float,
        max_grid_export_kw: float,
        p_diesel_kw: float = 0.0,
    ) -> PowerBalanceResult:
        """Resolve instantaneous AC bus balance and compute import, export, unserved load, and curtailment.

        Args:
            p_load_kw: Building electrical demand requested [kW >= 0].
            p_pv_ac_kw: Potential solar PV power available from inverter prior to bus-level curtailment [kW >= 0].
            p_batt_ch_ac_kw: Battery charging power absorbed from AC bus [kW >= 0].
            p_batt_dis_ac_kw: Battery discharging power injected into AC bus [kW >= 0].
            max_grid_import_kw: Contractual maximum grid import capacity [kW >= 0].
            max_grid_export_kw: Contractual maximum grid export capacity [kW >= 0].
            p_diesel_kw: Optional backup generator power [kW >= 0] (default: 0.0 kW).

        Returns:
            PowerBalanceResult dataclass containing resolved physical flows and residual.

        Raises:
            PhysicsViolationError: If any input power is negative or physically inconsistent.
            ConservationOfEnergyError: If power conservation is algebraically violated.
        """
        # Validate non-negativity of all inputs
        inputs = {
            "p_load_kw": p_load_kw,
            "p_pv_ac_kw": p_pv_ac_kw,
            "p_batt_ch_ac_kw": p_batt_ch_ac_kw,
            "p_batt_dis_ac_kw": p_batt_dis_ac_kw,
            "max_grid_import_kw": max_grid_import_kw,
            "max_grid_export_kw": max_grid_export_kw,
            "p_diesel_kw": p_diesel_kw,
        }
        for name, val in inputs.items():
            if val < 0.0:
                raise PhysicsViolationError(
                    f"Negative power quantity '{name}' = {val} kW violates physical non-negativity law",
                    details=inputs,
                )

        # Enforce battery mutual exclusivity on the AC bus
        if p_batt_ch_ac_kw > 0.0 and p_batt_dis_ac_kw > 0.0:
            raise PhysicsViolationError(
                f"Simultaneous battery charging ({p_batt_ch_ac_kw} kW) and discharging ({p_batt_dis_ac_kw} kW) violation"
            )

        # Potential net load to be served
        # P_net > 0 means demand exceeds local generation (deficit)
        # P_net < 0 means local generation exceeds demand (surplus)
        p_net = (p_load_kw + p_batt_ch_ac_kw) - (p_pv_ac_kw + p_batt_dis_ac_kw + p_diesel_kw)

        if p_net > 0.0:
            # DEFICIT: All available PV is delivered to the bus
            p_pv_delivered = p_pv_ac_kw
            p_pv_curtail = 0.0
            p_export = 0.0

            # Grid import supplies deficit up to contractual limit
            p_import = min(max_grid_import_kw, p_net)
            # Remaining unmet deficit cannot be served by grid
            p_unserved = max(0.0, p_net - max_grid_import_kw)
            p_served_load = p_load_kw - p_unserved

        elif p_net < 0.0:
            # SURPLUS: All load is served fully (zero unserved load)
            p_served_load = p_load_kw
            p_unserved = 0.0
            p_import = 0.0

            surplus = abs(p_net)
            # Grid export absorbs surplus up to contractual limit
            p_export = min(max_grid_export_kw, surplus)
            # Excess surplus that cannot be exported must be curtailed at the PV inverter
            p_pv_curtail = max(0.0, surplus - max_grid_export_kw)
            # Actual delivered PV to AC bus is reduced by the curtailed amount
            p_pv_delivered = p_pv_ac_kw - p_pv_curtail

        else:
            # EXACT BALANCE
            p_served_load = p_load_kw
            p_unserved = 0.0
            p_pv_delivered = p_pv_ac_kw
            p_pv_curtail = 0.0
            p_import = 0.0
            p_export = 0.0

        # Enforce exact Kirchhoff active power conservation:
        # P_sources = P_pv_delivered + P_grid_import + P_batt_dis_ac + P_diesel
        # P_sinks   = P_served_load + P_batt_ch_ac + P_grid_export
        sources = p_pv_delivered + p_batt_dis_ac_kw + p_import + p_diesel_kw
        sinks = p_served_load + p_batt_ch_ac_kw + p_export
        residual = abs(sources - sinks)

        if residual > self.tolerance_kw:
            raise ConservationOfEnergyError(
                f"Microgrid AC bus power conservation violated: residual {residual:.2e} kW exceeds tolerance {self.tolerance_kw:.2e} kW",
                details={
                    "sources_kw": sources,
                    "sinks_kw": sinks,
                    "residual_kw": residual,
                    "tolerance_kw": self.tolerance_kw,
                },
            )

        return PowerBalanceResult(
            p_pv_ac_delivered_kw=float(p_pv_delivered),
            p_grid_import_kw=float(p_import),
            p_batt_dis_ac_kw=float(p_batt_dis_ac_kw),
            p_diesel_kw=float(p_diesel_kw),
            p_served_load_kw=float(p_served_load),
            p_batt_ch_ac_kw=float(p_batt_ch_ac_kw),
            p_grid_export_kw=float(p_export),
            p_requested_load_kw=float(p_load_kw),
            p_unserved_load_kw=float(p_unserved),
            p_pv_ac_potential_kw=float(p_pv_ac_kw),
            p_pv_curtail_ac_kw=float(p_pv_curtail),
            residual_error_kw=float(residual),
            is_balanced=bool(residual <= self.tolerance_kw),
        )

"""Deterministic Rule-Based Dispatch Controller for microgrid_sim.

Governing Policy:
    1. Solar-First Demand Fulfillment:
       Available solar PV is prioritized to serve local building electrical demand.

    2. PV Surplus Absorbed by Battery:
       When PV generation exceeds building demand (P_pv > P_load):
       - If SoC < soc_max, dispatch battery to charge at min(P_pv - P_load, P_ch_max).
       - If SoC >= soc_max or battery cannot absorb all surplus, remaining power is
         exported to the grid up to contractual limits (or curtailed at inverter).

    3. PV Deficit Served by Battery:
       When building demand exceeds PV generation (P_load > P_pv):
       - If SoC > soc_min, dispatch battery to discharge at min(P_load - P_pv, P_dis_max).
       - If SoC <= soc_min or deficit exceeds battery capacity, remaining demand is
         supplied via utility grid import up to contractual limits.

    4. Invariant Protection:
       - Simultaneous charging and discharging is strictly avoided (a is a single scalar).
       - Battery SoC bounds [soc_min, soc_max] and power limits are strictly respected.
"""

from dataclasses import dataclass
from typing import Optional, Union
import numpy as np

from microgrid_sim.controllers.base import BaseController
from microgrid_sim.env.spaces import ObsIndex
from microgrid_sim.utils.logger import get_logger

logger = get_logger(__name__)


@dataclass
class RuleBasedControllerConfig:
    """Configurable parameters for rule-based dispatch controller."""
    soc_min: float = 0.15
    soc_max: float = 0.90
    min_power_deadband_kw: float = 0.05
    allow_grid_charging: bool = False
    allow_grid_export_discharge: bool = False


class RuleBasedController(BaseController):
    """Transparent, deterministic rule-based controller for microgrid energy dispatch."""

    def __init__(
        self,
        config: Optional[RuleBasedControllerConfig] = None,
        p_rated_ch_kw: float = 100.0,
        p_rated_dis_kw: float = 100.0,
        p_norm_load_kw: float = 100.0,
        p_norm_pv_kw: float = 120.0,
    ) -> None:
        """Initialize the rule-based controller.

        Args:
            config: Optional RuleBasedControllerConfig instance.
            p_rated_ch_kw: Maximum battery charging power rating [kW] (default: 100.0 kW).
            p_rated_dis_kw: Maximum battery discharging power rating [kW] (default: 100.0 kW).
            p_norm_load_kw: Load normalization factor used in environment observations [kW].
            p_norm_pv_kw: Solar PV normalization factor used in environment observations [kW].
        """
        self.config = config or RuleBasedControllerConfig()
        self.p_rated_ch_kw = float(p_rated_ch_kw)
        self.p_rated_dis_kw = float(p_rated_dis_kw)
        self.p_norm_load_kw = float(p_norm_load_kw)
        self.p_norm_pv_kw = float(p_norm_pv_kw)

        logger.debug(
            "Initialized RuleBasedController: P_ch=%.1f kW, P_dis=%.1f kW, SoC bounds=[%.2f, %.2f]",
            self.p_rated_ch_kw, self.p_rated_dis_kw, self.config.soc_min, self.config.soc_max
        )

    def compute_action(
        self,
        p_pv_kw: float,
        p_load_kw: float,
        soc: float,
    ) -> np.ndarray:
        """Compute dispatch action from raw physical state variables.

        Args:
            p_pv_kw: Available solar PV AC generation potential [kW >= 0].
            p_load_kw: Building active electrical demand [kW >= 0].
            soc: Battery state of charge [0.0, 1.0].

        Returns:
            1-dimensional numpy array with action a in [-1.0, 1.0].
                a > 0: Discharge battery at power a * P_dis_max
                a < 0: Charge battery at power |a| * P_ch_max
                a == 0: Battery idle
        """
        delta_p = float(p_pv_kw - p_load_kw)
        deadband = self.config.min_power_deadband_kw

        if delta_p > deadband:
            # SURPLUS SOLAR (P_pv > P_load):
            # Prioritize charging battery from green surplus
            if soc < self.config.soc_max:
                p_ch_desired = min(delta_p, self.p_rated_ch_kw)
                # Sign convention: charging request is negative
                action_val = - (p_ch_desired / self.p_rated_ch_kw) if self.p_rated_ch_kw > 0.0 else 0.0
            else:
                # Battery full: idle and let excess export to grid or curtail
                action_val = 0.0

        elif delta_p < -deadband:
            # DEFICIT SOLAR (P_load > P_pv):
            # Prioritize discharging battery to meet unmet load
            deficit = abs(delta_p)
            if soc > self.config.soc_min:
                p_dis_desired = min(deficit, self.p_rated_dis_kw)
                # Sign convention: discharging request is positive
                action_val = (p_dis_desired / self.p_rated_dis_kw) if self.p_rated_dis_kw > 0.0 else 0.0
            else:
                # Battery depleted: idle and let grid import meet unmet load
                action_val = 0.0

        else:
            # EXACT BALANCE (within deadband)
            action_val = 0.0

        action_clamped = float(np.clip(action_val, -1.0, 1.0))
        return np.array([action_clamped], dtype=np.float32)

    def act(self, observation: np.ndarray) -> np.ndarray:
        """Compute dispatch control action from Gymnasium observation vector.

        Args:
            observation: Environment observation vector conforming to create_observation_space().

        Returns:
            1-element action vector in [-1.0, 1.0].
        """
        soc = float(observation[ObsIndex.SOC])
        p_load = float(observation[ObsIndex.LOAD_DEMAND]) * self.p_norm_load_kw
        p_pv = float(observation[ObsIndex.PV_POTENTIAL]) * self.p_norm_pv_kw

        return self.compute_action(p_pv_kw=p_pv, p_load_kw=p_load, soc=soc)

    def reset(self) -> None:
        """Reset internal state between episodes (controller is stateless)."""
        pass

    plot_h = h - margin_t - margin_b

    x_min, x_max = float(x_vals[0]), float(x_vals[-1])
    if y_range is None:
        all_y = [val for s in series_list for val in s["y"]]
        y_min = min(0.0, min(all_y))
        y_max = max(all_y) * 1.1 if max(all_y) > 0 else 1.0
    else:
        y_min, y_max = y_range

    def get_x(val):
        return margin_l + (val - x_min) / (x_max - x_min) * plot_w

    def get_y(val):
        if y_max == y_min:
            return margin_t + plot_h / 2
        return margin_t + plot_h - (val - y_min) / (y_max - y_min) * plot_h

    svg = [
        f'<svg xmlns="http://www.w3.org/2000/svg" viewBox="0 0 {w} {h}" width="{w}" height="{h}" '
        f'style="background-color: #ffffff; font-family: -apple-system, BlinkMacSystemFont, Segoe UI, Roboto, sans-serif;">'
    ]

    # Title
    svg.append(f'<text x="{w/2}" y="32" font-size="18" font-weight="bold" text-anchor="middle" fill="#1e293b">{title}</text>')

    # Background grid & axes
    svg.append(f'<rect x="{margin_l}" y="{margin_t}" width="{plot_w}" height="{plot_h}" fill="#f8fafc" stroke="#cbd5e1" stroke-width="1"/>')

    # Horizontal grid lines (5 steps)
    y_ticks = np.linspace(y_min, y_max, 6)
    for yt in y_ticks:
        py = get_y(yt)
        svg.append(f'<line x1="{margin_l}" y1="{py}" x2="{margin_l + plot_w}" y2="{py}" stroke="#e2e8f0" stroke-width="1" stroke-dasharray="3,3"/>')
        svg.append(f'<text x="{margin_l - 10}" y="{py + 4}" font-size="11" text-anchor="end" fill="#64748b">{yt:.1f}</text>')

    # Vertical grid lines (every 3 hours)
    for x_tick in range(0, int(x_max) + 1, 3):
        px = get_x(x_tick)
        svg.append(f'<line x1="{px}" y1="{margin_t}" x2="{px}" y2="{margin_t + plot_h}" stroke="#e2e8f0" stroke-width="1" stroke-dasharray="3,3"/>')
        svg.append(f'<text x="{px}" y="{margin_t + plot_h + 18}" font-size="11" text-anchor="middle" fill="#64748b">{x_tick}:00</text>')

    # Axis Labels
    svg.append(f'<text x="{margin_l + plot_w / 2}" y="{h - 15}" font-size="13" font-weight="600" text-anchor="middle" fill="#334155">{x_label}</text>')
    svg.append(
        f'<text transform="rotate(-90)" x="{-margin_t - plot_h / 2}" y="24" font-size="13" font-weight="600" text-anchor="middle" fill="#334155">{y_label}</text>'
    )

    # Plot series
    legend_items = []
    for s in series_list:
        color = s["color"]
        name = s["name"]
        width = s.get("width", 2.5)
        dash = s.get("dash", "")
        dash_attr = f' stroke-dasharray="{dash}"' if dash else ""

        pts = [f"{get_x(x):.1f},{get_y(y):.1f}" for x, y in zip(x_vals, s["y"])]
        polyline = f'<polyline points="{" ".join(pts)}" fill="none" stroke="{color}" stroke-width="{width}"{dash_attr}/>'
        svg.append(polyline)

        # Fill under curve if requested
        if s.get("fill", False):
            fill_color = s.get("fill_color", color)
            opacity = s.get("fill_opacity", 0.15)
            pts_fill = [f"{get_x(x_min):.1f},{get_y(y_min):.1f}"] + pts + [f"{get_x(x_max):.1f},{get_y(y_min):.1f}"]
            svg.append(f'<polygon points="{" ".join(pts_fill)}" fill="{fill_color}" opacity="{opacity}"/>')

        legend_items.append((name, color, dash))

    # Optional annotations (e.g. bounds)
    if annotations:
        for ann in annotations:
            ann_y = get_y(ann["y"])
            svg.append(f'<line x1="{margin_l}" y1="{ann_y}" x2="{margin_l + plot_w}" y2="{ann_y}" stroke="{ann["color"]}" stroke-width="1.5" stroke-dasharray="4,4"/>')
            svg.append(f'<text x="{margin_l + plot_w - 5}" y="{ann_y - 5}" font-size="10" font-weight="bold" text-anchor="end" fill="{ann["color"]}">{ann["label"]}</text>')

    # Legend
    leg_x = margin_l + 20
    leg_y = margin_t + 20
    leg_w = len(legend_items) * 160 + 20
    svg.append(f'<rect x="{leg_x}" y="{leg_y}" width="{leg_w}" height="28" fill="#ffffff" stroke="#cbd5e1" stroke-width="1" rx="4" opacity="0.9"/>')
    for i, (name, col, dash) in enumerate(legend_items):
        item_x = leg_x + 15 + i * 160
        item_y = leg_y + 14
        dash_attr = ' stroke-dasharray="3,3"' if dash else ""
        svg.append(f'<line x1="{item_x}" y1="{item_y}" x2="{item_x + 25}" y2="{item_y}" stroke="{col}" stroke-width="2.5"{dash_attr}/>')
        svg.append(f'<text x="{item_x + 32}" y="{item_y + 4}" font-size="11" font-weight="500" fill="#334155">{name}</text>')

    svg.append("</svg>")
    with open(filepath, "w", encoding="utf-8") as f:
        f.write("\n".join(svg))


def _draw_pillow_png(
    filepath: Path,
    x_vals: np.ndarray,
    series_list: list[dict],
    title: str,
    x_label: str,
    y_label: str,
    y_range: tuple[float, float] = None,
    annotations: list[dict] = None,
) -> None:
    """Render crisp raster PNG using Pillow."""
    scale = 2  # Super-sample for crisp anti-aliasing
    w, h = 900 * scale, 480 * scale
    margin_l, margin_r, margin_t, margin_b = 80 * scale, 40 * scale, 60 * scale, 60 * scale
    plot_w = w - margin_l - margin_r
    plot_h = h - margin_t - margin_b

    im = Image.new("RGB", (w, h), color=(255, 255, 255))
    draw = ImageDraw.Draw(im)

    try:
        font_title = ImageFont.truetype("arial.ttf", 18 * scale)
        font_axis = ImageFont.truetype("arial.ttf", 13 * scale)
        font_ticks = ImageFont.truetype("arial.ttf", 10 * scale)
        font_leg = ImageFont.truetype("arial.ttf", 11 * scale)
    except OSError:
        font_title = font_axis = font_ticks = font_leg = ImageFont.load_default()

    x_min, x_max = float(x_vals[0]), float(x_vals[-1])
    if y_range is None:
        all_y = [val for s in series_list for val in s["y"]]
        y_min = min(0.0, min(all_y))
        y_max = max(all_y) * 1.1 if max(all_y) > 0 else 1.0
    else:
        y_min, y_max = y_range

    def get_x(val):
        return margin_l + (val - x_min) / (x_max - x_min) * plot_w

    def get_y(val):
        if y_max == y_min:
            return margin_t + plot_h / 2
        return margin_t + plot_h - (val - y_min) / (y_max - y_min) * plot_h

    # Title
    draw.text((w / 2, 24 * scale), title, fill=(30, 41, 59), font=font_title, anchor="mt")

    # Plot Background & Border
    draw.rectangle([margin_l, margin_t, margin_l + plot_w, margin_t + plot_h], fill=(248, 250, 252), outline=(203, 213, 225), width=scale)

    # Y grid lines
    y_ticks = np.linspace(y_min, y_max, 6)
    for yt in y_ticks:
        py = get_y(yt)
        draw.line([(margin_l, py), (margin_l + plot_w, py)], fill=(226, 232, 240), width=scale)
        draw.text((margin_l - 10 * scale, py), f"{yt:.1f}", fill=(100, 116, 139), font=font_ticks, anchor="rm")

    # X grid lines
    for x_tick in range(0, int(x_max) + 1, 3):
        px = get_x(x_tick)
        draw.line([(px, margin_t), (px, margin_t + plot_h)], fill=(226, 232, 240), width=scale)
        draw.text((px, margin_t + plot_h + 8 * scale), f"{x_tick}:00", fill=(100, 116, 139), font=font_ticks, anchor="mt")

    # Axis Labels
    draw.text((margin_l + plot_w / 2, h - 18 * scale), x_label, fill=(51, 65, 85), font=font_axis, anchor="mb")
    draw.text((25 * scale, margin_t + plot_h / 2), y_label, fill=(51, 65, 85), font=font_axis, anchor="mm")

    # Hex to RGB helper
    def hex_to_rgb(hex_str):
        hex_str = hex_str.lstrip("#")
        return tuple(int(hex_str[i:i+2], 16) for i in (0, 2, 4))

    # Plot Series
    for s in series_list:
        rgb = hex_to_rgb(s["color"])
        pts = [(get_x(x), get_y(y)) for x, y in zip(x_vals, s["y"])]
        draw.line(pts, fill=rgb, width=int(2.5 * scale))

    # Annotations
    if annotations:
        for ann in annotations:
            ann_rgb = hex_to_rgb(ann["color"])
            ann_py = get_y(ann["y"])
            draw.line([(margin_l, ann_py), (margin_l + plot_w, ann_py)], fill=ann_rgb, width=int(1.5 * scale))
            draw.text((margin_l + plot_w - 5 * scale, ann_py - 6 * scale), ann["label"], fill=ann_rgb, font=font_ticks, anchor="rb")

    # Legend
    leg_x = margin_l + 20 * scale
    leg_y = margin_t + 16 * scale
    leg_w = len(series_list) * 160 * scale + 20 * scale
    draw.rectangle([leg_x, leg_y, leg_x + leg_w, leg_y + 26 * scale], fill=(255, 255, 255), outline=(203, 213, 225), width=scale)
    for i, s in enumerate(series_list):
        item_x = leg_x + 15 * scale + i * 160 * scale
        item_y = leg_y + 13 * scale
        draw.line([(item_x, item_y), (item_x + 25 * scale, item_y)], fill=hex_to_rgb(s["color"]), width=int(2.5 * scale))
        draw.text((item_x + 30 * scale, item_y), s["name"], fill=(51, 65, 85), font=font_leg, anchor="lm")

    # Downsample for smooth anti-aliased output
    im_final = im.resize((900, 480), Image.Resampling.LANCZOS)
    im_final.save(filepath, "PNG")

) -> None:
    """Execute complete 24-hour closed-loop simulation of RuleBasedController in MicrogridEnv."""
    env = MicrogridEnv(config=complete_cfg, dataset=deterministic_24h_df)
    controller = RuleBasedController(
        p_rated_ch_kw=complete_cfg.battery.power_and_thermal.max_charge_power_kw,
        p_rated_dis_kw=complete_cfg.battery.power_and_thermal.max_discharge_power_kw,
    )

    obs, _ = env.reset(options={"initial_soc": 0.50})
    terminated = False
    step_count = 0

    while not terminated:
        action = controller.act(obs)
        obs, reward, terminated, truncated, info = env.step(action)
        step_count += 1

        # Check in-flight physical invariants:
        assert info["residual_error_kw"] < 1e-6
        assert info["is_balanced"] is True
        assert complete_cfg.battery.capacity.minimum_soc - 1e-6 <= info["soc"] <= complete_cfg.battery.capacity.maximum_soc + 1e-6

        # Check mutual exclusivity on battery AC bus
        assert not (info["p_batt_ch_ac_kw"] > 1e-6 and info["p_batt_dis_ac_kw"] > 1e-6)

    assert step_count == len(deterministic_24h_df)


def test_rule_based_controller_economic_outperformance(
    complete_cfg: CompleteSystemConfig,
    deterministic_24h_df: pd.DataFrame,
) -> None:
    """Verify that rule-based battery dispatch reduces total grid electricity cost vs passive baseline."""
    # 1. Run Rule-Based Controlled Microgrid
    env_rb = MicrogridEnv(config=complete_cfg, dataset=deterministic_24h_df)
    controller = RuleBasedController()

    obs_rb, _ = env_rb.reset(options={"initial_soc": 0.50})
    term_rb = False
    while not term_rb:
        act = controller.act(obs_rb)
        obs_rb, _, term_rb, _, _ = env_rb.step(act)
    cost_rb = env_rb.cumulative_cost_usd

    # 2. Run Passive (Zero-Battery / Idle) Microgrid
    env_passive = MicrogridEnv(config=complete_cfg, dataset=deterministic_24h_df)
    obs_p, _ = env_passive.reset(options={"initial_soc": 0.50})
    term_p = False
    while not term_p:
        # Action is always 0.0 (no battery dispatch)
        obs_p, _, term_p, _, _ = env_passive.step(np.array([0.0], dtype=np.float32))
    cost_passive = env_passive.cumulative_cost_usd

    # Active battery dispatch stores cheap daytime surplus solar and offsets expensive grid purchases,
    # thereby lowering net operating cost compared to a passive installation.
    assert cost_rb < cost_passive
    savings_pct = (cost_passive - cost_rb) / cost_passive * 100.0
    assert savings_pct > 5.0  # Demonstrates meaningful economic benefit
